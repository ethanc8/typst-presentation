\documentclass{article}
\usepackage{luacode}
\begin{luacode*}
  function sieve_lua(n)
    local is_prime = {}
    for i = 2, n do is_prime[i] = true end
    for i = 2, math.sqrt(n) do
      if is_prime[i] then
        for j = i * i, n, i do
          is_prime[j] = false
        end
      end
    end
    local results = {}
    for i = 2, n do
      if is_prime[i] then
        table.insert(results, i)
      end
    end
    tex.print(table.concat(results, ", "))
  end
\end{luacode*}
\newcommand{\sieve}[1]{\directlua{sieve_lua(#1)}}

\begin{document}
\sieve{50}
\end{document}
